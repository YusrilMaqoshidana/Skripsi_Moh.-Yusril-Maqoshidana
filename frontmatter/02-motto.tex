\frontmatterchapter{MOTTO}

\begin{center}

    \textit{``
    Aku telah berupaya tanpa henti untuk tidak mencemooh, tidak meratapi, tidak menghina tindakan manusia, tetapi untuk memahaminya.
    ''} \\
    (Baruch Spinoza)

    \vspace{24pt}

    \textit{
        ``
        Your worst sin is that you have destroyed and betrayed yourself for nothing.
        ''
    } \\
    (Fyodor Dostoevsky) \\

\end{center}

\frontmatterchapter{PERSEMBAHAN}

% --- ISI PERSEMBAHAN ---
% Ganti konten lipsum berikut dengan teks persembahan Anda
\noindent Halaman ini saya persembahkan kepada:

\begin{enumerate}[nosep]
    \item Allah SWT, atas segala rahmat, karunia, serta kemudahan yang diberikan sehingga penulis dapat menyelesaikan skripsi ini dengan baik.

    \item Orang tua tercinta, atas doa, kasih sayang, dukungan, dan segala pengorbanan yang telah diberikan sehingga penulis dapat menyelesaikan pendidikan sarjana ini.

    \item Bapak \supervisor\ selaku Dosen Pembimbing Utama yang telah meluangkan waktu, memberikan arahan, bimbingan, serta masukan yang berharga selama proses penyusunan skripsi ini.

    \item Seluruh dosen yang telah membagikan ilmu, wawasan, dan pengalaman selama masa perkuliahan sehingga menjadi bekal dalam penyusunan skripsi ini.

    \item Sahabat, teman seperjuangan, dan seluruh pihak yang telah memberikan dukungan, semangat, serta bantuan selama proses penyusunan skripsi ini.
    
    \item Kepada diri sendiri, atas perjuangan dalam menghadapi berbagai rintangan selama proses penyusunan skripsi ini.
\end{enumerate}

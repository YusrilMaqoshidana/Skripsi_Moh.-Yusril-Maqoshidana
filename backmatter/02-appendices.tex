\subsection{Lampiran B.1 - Kuesioner}
\label{subsec:appx-questionnaire}

\noindent \textbf{KUESIONER PENELITIAN}

Petunjuk Pengisian:
\begin{itemize}[nosep]
    \item Bacalah setiap pertanyaan dengan cermat
    \item Jawab dengan jujur sesuai pendapat Anda
    \item Tidak ada jawaban yang benar atau salah
    \item Data Anda dijamin kerahasiaannya
\end{itemize}

\textbf{A. DATA DEMOGRAFIS}

\begin{enumerate}
    \item Jenis Kelamin: Laki-laki $\square$ \quad Perempuan $\square$
    \item Usia: ... tahun
    \item Pendidikan: ...
    \item Pekerjaan: ...
\end{enumerate}

\textbf{B. PERTANYAAN PENELITIAN}

Untuk pertanyaan berikut, berikan jawaban dengan skala 1-5:
\begin{itemize}[nosep]
    \item 1 = Sangat Tidak Setuju (STS)
    \item 2 = Tidak Setuju (TS)
    \item 3 = Netral (N)
    \item 4 = Setuju (S)
    \item 5 = Sangat Setuju (SS)
\end{itemize}

\begin{enumerate}
    \item Pernyataan pertama: $\quad$ STS $\square$ TS $\square$ N $\square$ S $\square$ SS $\square$
    \item Pernyataan kedua: $\quad$ STS $\square$ TS $\square$ N $\square$ S $\square$ SS $\square$
    \item Pernyataan ketiga: $\quad$ STS $\square$ TS $\square$ N $\square$ S $\square$ SS $\square$
\end{enumerate}
